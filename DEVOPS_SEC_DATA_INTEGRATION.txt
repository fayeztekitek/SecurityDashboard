STEP 2.3: Web Scraping (If no API/Export)
   
   If no direct export, use Python web scraping:
   
```python
#!/usr/bin/env python3
"""
DevOps-Sec Portal Data Extractor
Scrapes vulnerability data from devops-sec.vermeg.com
"""

import requests
from bs4 import BeautifulSoup
import csv
import re
from datetime import datetime

class DevopsSecExtractor:
    def __init__(self):
        self.base_url = "https://devops-sec.vermeg.com"
        self.session = requests.Session()
        self.session.headers.update({
            'User-Agent': 'Mozilla/5.0 (Security Dashboard Collector)'
        })
    
    def get_vulnerabilities(self):
        """Fetch vulnerabilities from portal"""
        try:
            print("[*] Fetching vulnerabilities from devops-sec...")
            resp = self.session.get(f"{self.base_url}/vulnerabilities")
            resp.raise_for_status()
            
            soup = BeautifulSoup(resp.content, 'html.parser')
            
            vulns = []
            
            # Find vulnerability tables or cards
            # Adjust selectors based on actual page structure
            vuln_rows = soup.find_all('tr', class_='vulnerability-row')
            
            for i, row in enumerate(vuln_rows, 1):
                try:
                    # Extract data from table cells
                    cells = row.find_all('td')
                    
                    vuln = {
                        'Vuln_ID': f"VUL-{datetime.now().year}-{i:04d}",
                        'CVE': self._extract_text(cells, 1),  # Adjust index
                        'Severity': self._extract_text(cells, 2),
                        'Product': self._extract_text(cells, 3),
                        'Component': self._extract_text(cells, 4),
                        'Discovery_Date': self._extract_date(self._extract_text(cells, 5)),
                        'Days_Overdue': self._calculate_overdue(self._extract_text(cells, 5)),
                        'Owner': self._extract_text(cells, 6),
                        'Status': self._extract_text(cells, 7),
                        'Remediation_Plan': self._extract_text(cells, 8)
                    }
                    
                    vulns.append(vuln)
                    print(f"[+] Extracted: {vuln['Vuln_ID']} ({vuln['Severity']})")
                
                except Exception as e:
                    print(f"[-] Error parsing row: {e}")
                    continue
            
            return vulns
        
        except Exception as e:
            print(f"[-] Error fetching vulnerabilities: {e}")
            return []
    
    def _extract_text(self, cells, index):
        """Safely extract text from cell"""
        try:
            return cells[index].get_text(strip=True)
        except:
            return ""
    
    def _extract_date(self, date_str):
        """Parse and format date"""
        try:
            # Try multiple date formats
            for fmt in ['%Y-%m-%d', '%m/%d/%Y', '%d-%m-%Y']:
                try:
                    parsed = datetime.strptime(date_str, fmt)
                    return parsed.strftime('%Y-%m-%d')
                except:
                    continue
            return datetime.now().strftime('%Y-%m-%d')
        except:
            return datetime.now().strftime('%Y-%m-%d')
    
    def _calculate_overdue(self, date_str):
        """Calculate days overdue from date"""
        try:
            discovery_date = datetime.strptime(date_str, '%Y-%m-%d')
            days_old = (datetime.now() - discovery_date).days
            
            # SLA: 30 days for critical
            sla_days = 30
            overdue = max(0, days_old - sla_days)
            
            return str(overdue)
        except:
            return "0"
    
    def save_to_csv(self, vulns, filename='devops_sec_vulnerabilities.csv'):
        """Save vulnerabilities to CSV"""
        if not vulns:
            print("[-] No vulnerabilities to save")
            return False
        
        fieldnames = ['Vuln_ID', 'CVE', 'Severity', 'Product', 'Component',
                      'Discovery_Date', 'Days_Overdue', 'Owner', 'Status', 'Remediation_Plan']
        
        try:
            with open(filename, 'w', newline='', encoding='utf-8') as f:
                writer = csv.DictWriter(f, fieldnames=fieldnames)
                writer.writeheader()
                writer.writerows(vulns)
            
            print(f"[✓] Saved {len(vulns)} vulnerabilities to {filename}")
            return True
        
        except Exception as e:
            print(f"[-] Error saving to CSV: {e}")
            return False

def main():
    print("=" * 60)
    print("DevOps-Sec Portal Data Extractor")
    print("=" * 60)
    
    extractor = DevopsSecExtractor()
    
    print("\n[*] Extracting data from devops-sec.vermeg.com...")
    vulns = extractor.get_vulnerabilities()
    
    if vulns:
        print(f"\n[✓] Successfully extracted {len(vulns)} vulnerabilities")
        extractor.save_to_csv(vulns)
        print("\n[✓] Data ready for dashboard import!")
    else:
        print("[-] No vulnerabilities extracted")

if __name__ == '__main__':
    main()
```

STEP 3.1: Handle Different Data Sources
   
   DevOps-Sec typically aggregates from:
   - Fortify SSC
   - Sonatype Nexus IQ
   - Container scanners
   - SAST/DAST tools
   
   Ensure your mapping handles all sources

STEP 3.2: Data Standardization
   
   Different sources may use different fields:
   
   Source              Severity Field      Product Field
   ─────────────────────────────────────────────────────
   Fortify             "severity"          "application"
   Nexus IQ            "threatLevel"       "application"
   Container Scan      "severity"          "image"
   SAST Tool          "level"              "project"
   
   Create mapping:
   ```python
   severity_mapping = {
       'CRITICAL': 'Critical',
       'Critical': 'Critical',
       'HIGH': 'High',
       'High': 'High',
       'MEDIUM': 'Medium',
       'Medium': 'Medium',
       'LOW': 'Low',
       'Low': 'Low'
   }
   
   product_mapping = {
       'veggo-banking-platform': 'Veggo',
       'megara-core-system': 'Megara',
       # ... etc
   }
   ```

STEP 3.3: Deduplication
   
   DevOps-Sec may show same vulnerability multiple times
   if found by different scanners:
   
   ```python
   # Keep only one instance of each CVE per product
   seen = set()
   unique_vulns = []
   
   for vuln in vulns:
       key = (vuln['CVE'], vuln['Product'])
       if key not in seen:
           seen.add(key)
           unique_vulns.append(vuln)
   ```

STEP 3.4: Validation
   
   Before importing:
   ```python
   def validate_vuln(vuln):
       errors = []
       
       if not vuln.get('Vuln_ID'):
           errors.append("Missing Vuln_ID")
       if vuln.get('Severity') not in ['Critical', 'High', 'Medium', 'Low']:
           errors.append(f"Invalid Severity: {vuln.get('Severity')}")
       if not vuln.get('Product'):
           errors.append("Missing Product")
       
       return errors
   ```

```bash
#!/bin/bash
# devops_sec_extract.sh - Daily vulnerability extraction

TIMESTAMP=$(date +%Y-%m-%d)
```

STEP 5: COMBINE ALL DATA SOURCES
================================================================================

Final Integration with Fortify + Nexus IQ + DevOps-Sec:

```python
#!/usr/bin/env python3
"""
Merge data from all three sources into single dashboard CSV
"""

import pandas as pd
from datetime import datetime

# Load data from each source
fortify_df = pd.read_csv('fortify_vulnerabilities.csv')
nexus_df = pd.read_csv('nexus_iq_vulnerabilities.csv')
devops_df = pd.read_csv('devops_sec_vulnerabilities.csv')

print(f"Fortify: {len(fortify_df)} vulnerabilities")
print(f"Nexus IQ: {len(nexus_df)} vulnerabilities")
print(f"DevOps-Sec: {len(devops_df)} vulnerabilities")

# Combine all
combined_df = pd.concat([fortify_df, nexus_df, devops_df], ignore_index=True)

# Remove duplicates (keep first occurrence)
combined_df = combined_df.drop_duplicates(subset=['CVE', 'Product'], keep='first')

# Ensure data quality
combined_df = combined_df[combined_df['Severity'].isin(['Critical', 'High', 'Medium', 'Low'])]
combined_df = combined_df.dropna(subset=['Vuln_ID', 'Product'])

# Sort by severity (Critical first) and days overdue
severity_order = {'Critical': 0, 'High': 1, 'Medium': 2, 'Low': 3}
combined_df['severity_order'] = combined_df['Severity'].map(severity_order)
combined_df['Days_Overdue'] = pd.to_numeric(combined_df['Days_Overdue'], errors='coerce').fillna(0)
combined_df = combined_df.sort_values(['severity_order', 'Days_Overdue'], ascending=[True, False])

# Save final CSV
output_file = f"dashboard_vulnerabilities_{datetime.now().strftime('%Y-%m-%d')}.csv"
combined_df.to_csv(output_file, index=False)

print(f"\n✓ Combined dataset: {len(combined_df)} unique vulnerabilities")
print(f"✓ Saved to: {output_file}")
```

```bash
python3 combine_datasources.py
```

================================================================================
COMMON ISSUES & SOLUTIONS
================================================================================

ISSUE 1: DevOps-Sec requires login
SOLUTION:
   - Add authentication to script:
   ```python
   session.post(
       'https://devops-sec.vermeg.com/login',
       data={'username': user, 'password': pwd}
   )
   ```

ISSUE 2: Data format differences across sources
SOLUTION:
   - Map all sources to consistent format
   - Use config file for mappings
   - Validate before combining

ISSUE 3: Duplicate vulnerabilities
SOLUTION:
   - Deduplicate on CVE + Product
   - Keep most recent/authoritative source
   - Add source attribution field

ISSUE 4: Performance (large datasets)
SOLUTION:
   - Use pandas for efficiency
   - Process in chunks
   - Archive old data
   - Implement incremental updates
