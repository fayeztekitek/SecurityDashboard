STEP 3: CREATE MASTER EXTRACTION SCRIPT
================================================================================

```python
#!/usr/bin/env python3
"""
Master Data Extractor
Pulls vulnerability data from all three sources
"""

import json
import subprocess
import os
from datetime import datetime
import logging

# Configure logging
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(levelname)s - %(message)s',
    handlers=[
        logging.FileHandler('logs/extraction.log'),
        logging.StreamHandler()
    ]
)
logger = logging.getLogger(__name__)

class MasterExtractor:
    def __init__(self):
        self.config_dir = 'config'
        self.data_raw_dir = 'data/raw'
        self.data_processed_dir = 'data/processed'
        self.timestamp = datetime.now().strftime('%Y-%m-%d_%H-%M-%S')
        
    def extract_all(self):
        """Extract from all sources"""
        logger.info("=" * 60)
        logger.info("Starting master vulnerability extraction")
        logger.info("=" * 60)
        
        results = {}
        
        # Extract Fortify
        logger.info("\n[1/3] Extracting Fortify data...")
        try:
            result = self._extract_fortify()
            results['fortify'] = result
            logger.info(f"✓ Fortify: {result['count']} vulnerabilities")
        except Exception as e:
            logger.error(f"✗ Fortify extraction failed: {e}")
            results['fortify'] = {'success': False, 'error': str(e)}
        
        # Extract Nexus IQ
        logger.info("\n[2/3] Extracting Nexus IQ data...")
        try:
            result = self._extract_nexus_iq()
            results['nexus_iq'] = result
            logger.info(f"✓ Nexus IQ: {result['count']} vulnerabilities")
        except Exception as e:
            logger.error(f"✗ Nexus IQ extraction failed: {e}")
            results['nexus_iq'] = {'success': False, 'error': str(e)}
        
        # Extract DevOps-Sec
        logger.info("\n[3/3] Extracting DevOps-Sec data...")
        try:
            result = self._extract_devops_sec()
            results['devops_sec'] = result
            logger.info(f"✓ DevOps-Sec: {result['count']} vulnerabilities")
        except Exception as e:
            logger.error(f"✗ DevOps-Sec extraction failed: {e}")
            results['devops_sec'] = {'success': False, 'error': str(e)}
        
        return results
    
    def _extract_fortify(self):
        """Run Fortify extraction script"""
        logger.info("Running Fortify extraction script...")
        subprocess.run(['python3', 'scripts/extract_fortify.py'])
        
        return {
            'success': True,
            'file': f'{self.data_raw_dir}/fortify_raw.csv',
            'count': self._count_csv_rows(f'{self.data_raw_dir}/fortify_raw.csv')
        }
    
    def _extract_nexus_iq(self):
        """Run Nexus IQ extraction script"""
        logger.info("Running Nexus IQ extraction script...")
        subprocess.run(['python3', 'scripts/extract_nexus_iq.py'])
        
        return {
            'success': True,
            'file': f'{self.data_raw_dir}/nexus_raw.csv',
            'count': self._count_csv_rows(f'{self.data_raw_dir}/nexus_raw.csv')
        }
    
    def _extract_devops_sec(self):
        """Run DevOps-Sec extraction script"""
        logger.info("Running DevOps-Sec extraction script...")
        subprocess.run(['python3', 'scripts/extract_devops_sec.py'])
        
        return {
            'success': True,
            'file': f'{self.data_raw_dir}/devops_raw.csv',
            'count': self._count_csv_rows(f'{self.data_raw_dir}/devops_raw.csv')
        }
    
    def _count_csv_rows(self, filepath):
        """Count rows in CSV file"""
        try:
            with open(filepath) as f:
                return sum(1 for _ in f) - 1  # Subtract header
        except:
            return 0

def main():
    extractor = MasterExtractor()
    results = extractor.extract_all()
    
    # Summary
    logger.info("\n" + "=" * 60)
    logger.info("EXTRACTION SUMMARY")
    logger.info("=" * 60)
    
    total = sum([r.get('count', 0) for r in results.values() if r.get('success')])
    logger.info(f"Total vulnerabilities extracted: {total}")
    
    for source, result in results.items():
        if result.get('success'):
            logger.info(f"✓ {source}: {result['count']} vulnerabilities")
        else:
            logger.info(f"✗ {source}: {result.get('error', 'Unknown error')}")
    
    logger.info("\nNext step: Run combine_datasources.py")

if __name__ == '__main__':
    main()
```

STEP 4: CREATE DATA COMBINATION/DEDUPLICATION SCRIPT
================================================================================

```python
#!/usr/bin/env python3
"""
Combine vulnerability data from all three sources
Deduplicate and produce single authoritative dataset
"""

import pandas as pd
import json
from datetime import datetime
import logging

logger = logging.getLogger(__name__)

class DataCombiner:
    def __init__(self):
        self.fortify_file = 'data/raw/fortify_raw.csv'
        self.nexus_file = 'data/raw/nexus_raw.csv'
        self.devops_file = 'data/raw/devops_raw.csv'
        
    def combine(self):
        """Combine all three data sources"""
        logger.info("Combining data from all sources...")
        
        # Load each source
        fortify_df = pd.read_csv(self.fortify_file)
        nexus_df = pd.read_csv(self.nexus_file)
        devops_df = pd.read_csv(self.devops_file)
        
        logger.info(f"Fortify: {len(fortify_df)} rows")
        logger.info(f"Nexus IQ: {len(nexus_df)} rows")
        logger.info(f"DevOps-Sec: {len(devops_df)} rows")
        
        # Add source tracking
        fortify_df['source'] = 'Fortify SSC'
        nexus_df['source'] = 'Nexus IQ'
        devops_df['source'] = 'DevOps-Sec'
        
        # Combine
        combined = pd.concat([fortify_df, nexus_df, devops_df], ignore_index=True)
        logger.info(f"Combined (before dedup): {len(combined)} rows")
        
        # Deduplicate by CVE + Product (keep first occurrence)
        combined = combined.drop_duplicates(
            subset=['CVE', 'Product'],
            keep='first'
        )
        logger.info(f"After deduplication: {len(combined)} unique vulnerabilities")
        
        # Validate data
        combined = self._validate_data(combined)
        logger.info(f"After validation: {len(combined)} rows")
        
        # Sort by priority
        combined = self._sort_by_priority(combined)
        
        # Save
        output_file = f"data/processed/dashboard_vulnerabilities_{datetime.now().strftime('%Y-%m-%d')}.csv"
        combined.to_csv(output_file, index=False)
        logger.info(f"✓ Saved combined data to {output_file}")
        
        return combined
    
    def _validate_data(self, df):
        """Validate and clean data"""
        # Ensure valid severity
        valid_severities = ['Critical', 'High', 'Medium', 'Low']
        df = df[df['Severity'].isin(valid_severities)]
        
        # Remove rows missing required fields
        required_fields = ['Vuln_ID', 'Product', 'Severity']
        df = df.dropna(subset=required_fields)
        
        return df
    
    def _sort_by_priority(self, df):
        """Sort by severity and days overdue"""
        severity_order = {'Critical': 0, 'High': 1, 'Medium': 2, 'Low': 3}
        df['severity_sort'] = df['Severity'].map(severity_order)
        df['Days_Overdue'] = pd.to_numeric(df['Days_Overdue'], errors='coerce').fillna(0)
        
        df = df.sort_values(
            ['severity_sort', 'Days_Overdue'],
            ascending=[True, False]
        )
        
        return df.drop('severity_sort', axis=1)
    
    def generate_summary(self, combined_df):
        """Generate summary statistics"""
        summary = {
            'timestamp': datetime.now().isoformat(),
            'total_vulnerabilities': len(combined_df),
            'by_severity': combined_df['Severity'].value_counts().to_dict(),
            'by_product': combined_df['Product'].value_counts().to_dict(),
            'by_source': combined_df['source'].value_counts().to_dict(),
            'overdue_count': len(combined_df[combined_df['Days_Overdue'] > 0])
        }
        
        return summary

def main():
    combiner = DataCombiner()
    combined = combiner.combine()
    summary = combiner.generate_summary(combined)
    
    print("\n" + "=" * 60)
    print("COMBINATION SUMMARY")
    print("=" * 60)
    print(f"Total vulnerabilities: {summary['total_vulnerabilities']}")
    print(f"\nBy Severity:")
    for sev, count in summary['by_severity'].items():
        print(f"  {sev}: {count}")
    print(f"\nBy Source:")
    for src, count in summary['by_source'].items():
        print(f"  {src}: {count}")
    print(f"\nOverdue: {summary['overdue_count']}")

if __name__ == '__main__':
    main()
```

================================================================================
STEP 5: CREATE SCHEDULED PIPELINE
================================================================================

Create cron job (runs daily at 6:00 AM):

```bash
#!/bin/bash
# /opt/security-dashboard/daily_extraction.sh

set -e

LOG_FILE="/var/log/security-dashboard/extraction_$(date +%Y-%m-%d).log"
SCRIPT_DIR="/opt/security-dashboard"

echo "Starting daily vulnerability extraction at $(date)" >> $LOG_FILE

cd $SCRIPT_DIR

# Step 1: Extract from all sources
echo "Step 1: Extracting data..." >> $LOG_FILE
python3 scripts/master_extractor.py >> $LOG_FILE 2>&1

# Step 2: Combine and deduplicate
echo "Step 2: Combining data..." >> $LOG_FILE
python3 scripts/combine_datasources.py >> $LOG_FILE 2>&1

# Step 3: Validate quality
echo "Step 3: Validating data..." >> $LOG_FILE
python3 scripts/validate_data.py >> $LOG_FILE 2>&1

# Step 4: Copy to dashboard location
echo "Step 4: Copying to dashboard..." >> $LOG_FILE
cp data/processed/dashboard_vulnerabilities_*.csv /var/www/dashboards/
chown www-data:www-data /var/www/dashboards/dashboard_vulnerabilities*.csv

# Step 5: Generate reports
echo "Step 5: Generating reports..." >> $LOG_FILE
python3 scripts/generate_reports.py >> $LOG_FILE 2>&1

echo "Extraction complete at $(date)" >> $LOG_FILE
```

Add to crontab:
```bash
crontab -e
# Add line:
0 6 * * * /opt/security-dashboard/daily_extraction.sh
```
